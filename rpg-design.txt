def _safe_dict(value: Any) -> Dict[str, Any]:
    return value if isinstance(value, dict) else {}


def _safe_list(value: Any) -> List[Any]:
    return value if isinstance(value, list) else []


def _safe_str(value: Any) -> str:
    return "" if value is None else str(value)


def _safe_bool(value: Any, default: bool = False) -> bool:
    if isinstance(value, bool):
        return value
    if value is None:
        return default
    text = str(value).strip().lower()
    if text in {"1", "true", "yes", "y", "on"}:
        return True
    if text in {"0", "false", "no", "n", "off"}:
        return False
    return default

def normalize_area(area: Any) -> Dict[str, Any]:
    area = dict(_safe_dict(area))
    area_id = _safe_str(area.get("area_id") or area.get("id")).strip()
    if area_id:
        area["area_id"] = area_id
    area["name"] = _safe_str(area.get("name") or area_id).strip()
    area["exits"] = [_safe_str(x).strip() for x in _safe_list(area.get("exits")) if _safe_str(x).strip()]
    area["occupants"] = [_safe_str(x).strip() for x in _safe_list(area.get("occupants")) if _safe_str(x).strip()]
    area["objects"] = [_safe_str(x).strip() for x in _safe_list(area.get("objects")) if _safe_str(x).strip()]
    return area


def normalize_barrier(barrier: Any) -> Dict[str, Any]:
    barrier = dict(_safe_dict(barrier))
    barrier_id = _safe_str(barrier.get("barrier_id") or barrier.get("id")).strip()
    if barrier_id:
        barrier["barrier_id"] = barrier_id
    between = [_safe_str(x).strip() for x in _safe_list(barrier.get("between")) if _safe_str(x).strip()]
    barrier["between"] = between[:2]
    barrier["door_object_id"] = _safe_str(barrier.get("door_object_id")).strip()
    barrier["locked"] = _safe_bool(barrier.get("locked"), False)
    barrier["open"] = _safe_bool(barrier.get("open"), False)
    barrier["blocks_movement"] = _safe_bool(barrier.get("blocks_movement"), False)
    barrier["blocks_touch"] = _safe_bool(barrier.get("blocks_touch"), False)
    barrier["blocks_visibility"] = _safe_bool(barrier.get("blocks_visibility"), False)
    barrier["muffles_sound"] = _safe_bool(barrier.get("muffles_sound"), False)
    barrier["blocks_whisper"] = _safe_bool(barrier.get("blocks_whisper"), False)
    barrier["allows_yell"] = _safe_bool(barrier.get("allows_yell"), True)
    return barrier


def normalize_scene_graph(value: Any) -> Dict[str, Any]:
    value = dict(_safe_dict(value))
    if not value:
        value = deepcopy(DEFAULT_SCENE_GRAPH)

    areas_raw = _safe_dict(value.get("areas"))
    areas: Dict[str, Dict[str, Any]] = {}
    for area_id, area in areas_raw.items():
        area = normalize_area(area)
        aid = _safe_str(area.get("area_id") or area_id).strip()
        if aid:
            areas[aid] = area

    barriers_raw = _safe_dict(value.get("barriers"))
    barriers: Dict[str, Dict[str, Any]] = {}
    for barrier_id, barrier in barriers_raw.items():
        barrier = normalize_barrier(barrier)
        bid = _safe_str(barrier.get("barrier_id") or barrier_id).strip()
        if bid:
            barriers[bid] = barrier

    actor_area = {}
    for actor_id, area_id in _safe_dict(value.get("actor_area")).items():
        actor_id = _safe_str(actor_id).strip()
        area_id = _safe_str(area_id).strip()
        if actor_id and area_id:
            actor_area[actor_id] = area_id

    # Backfill actor_area from area occupants.
    for area_id, area in areas.items():
        for actor_id in _safe_list(area.get("occupants")):
            actor_area.setdefault(_safe_str(actor_id), area_id)

    object_area = {}
    for area_id, area in areas.items():
        for object_id in _safe_list(area.get("objects")):
            object_area[_safe_str(object_id)] = area_id

    object_area.update({
        _safe_str(k): _safe_str(v)
        for k, v in _safe_dict(value.get("object_area")).items()
        if _safe_str(k) and _safe_str(v)
    })

    return {
        "location_id": _safe_str(value.get("location_id") or "loc_rusty_flagon").strip(),
        "areas": areas,
        "barriers": barriers,
        "actor_area": actor_area,
        "object_area": object_area,
    }


def scene_graph_from_state(simulation_state: Dict[str, Any]) -> Dict[str, Any]:
    simulation_state = _safe_dict(simulation_state)
    graph = (
        simulation_state.get("scene_graph")
        or _safe_dict(simulation_state.get("spatial_state")).get("scene_graph")
        or _safe_dict(simulation_state.get("interaction_state")).get("scene_graph")
    )
    return normalize_scene_graph(graph or DEFAULT_SCENE_GRAPH)


def set_scene_graph(simulation_state: Dict[str, Any], scene_graph: Dict[str, Any]) -> Dict[str, Any]:
    simulation_state = dict(_safe_dict(simulation_state))
    simulation_state["scene_graph"] = normalize_scene_graph(scene_graph)
    return simulation_state


def actor_area(scene_graph: Dict[str, Any], actor_id: str) -> str:
    return _safe_str(_safe_dict(scene_graph.get("actor_area")).get(_safe_str(actor_id).strip())).strip()


def object_area(scene_graph: Dict[str, Any], object_id: str) -> str:
    return _safe_str(_safe_dict(scene_graph.get("object_area")).get(_safe_str(object_id).strip())).strip()


def barrier_between(scene_graph: Dict[str, Any], area_a: str, area_b: str) -> Dict[str, Any]:
    area_a = _safe_str(area_a).strip()
    area_b = _safe_str(area_b).strip()
    if not area_a or not area_b:
        return {}
    wanted = {area_a, area_b}
    for barrier in _safe_dict(scene_graph.get("barriers")).values():
        between = set(_safe_list(_safe_dict(barrier).get("between")))
        if between == wanted:
            return _safe_dict(barrier)
    return {}


def areas_adjacent(scene_graph: Dict[str, Any], area_a: str, area_b: str) -> bool:
    if area_a == area_b:
        return True
    area = _safe_dict(_safe_dict(scene_graph.get("areas")).get(area_a))
    return area_b in set(_safe_list(area.get("exits")))


def can_move_between(scene_graph: Dict[str, Any], from_area: str, to_area: str) -> Tuple[bool, str, Dict[str, Any]]:
    if not from_area or not to_area:
        return False, "missing_area", {}
    if from_area == to_area:
        return True, "same_area", {}
    if not areas_adjacent(scene_graph, from_area, to_area):
        return False, "areas_not_connected", {}

    barrier = barrier_between(scene_graph, from_area, to_area)
    if barrier:
        if barrier.get("locked"):
            return False, "barrier_locked", barrier
        if barrier.get("blocks_movement") and not barrier.get("open"):
            return False, "barrier_closed", barrier

    return True, "movement_allowed", barrier


def can_reach_object(scene_graph: Dict[str, Any], actor_id: str, object_id: str) -> Tuple[bool, str, Dict[str, Any]]:
    a_area = actor_area(scene_graph, actor_id)
    o_area = object_area(scene_graph, object_id)
    if not a_area or not o_area:
        return False, "missing_spatial_reference", {}
    if a_area == o_area:
        return True, "same_area", {}

    allowed, reason, barrier = can_move_between(scene_graph, a_area, o_area)
    if not allowed:
        return False, reason, barrier
    return True, "reachable_adjacent_area", barrier


def can_touch_actor(scene_graph: Dict[str, Any], actor_id: str, target_actor_id: str) -> Tuple[bool, str, Dict[str, Any]]:
    a_area = actor_area(scene_graph, actor_id)
    t_area = actor_area(scene_graph, target_actor_id)
    if not a_area or not t_area:
        return False, "missing_spatial_reference", {}
    if a_area == t_area:
        return True, "same_area", {}

    barrier = barrier_between(scene_graph, a_area, t_area)
    if barrier and (barrier.get("blocks_touch") or barrier.get("locked") or not barrier.get("open")):
        return False, "barrier_blocks_touch", barrier

    return False, "not_same_area", barrier


def can_see_actor(scene_graph: Dict[str, Any], actor_id: str, target_actor_id: str) -> Tuple[bool, str, Dict[str, Any]]:
    a_area = actor_area(scene_graph, actor_id)
    t_area = actor_area(scene_graph, target_actor_id)
    if not a_area or not t_area:
        return False, "missing_spatial_reference", {}
    if a_area == t_area:
        return True, "same_area", {}

    barrier = barrier_between(scene_graph, a_area, t_area)
    if barrier and (barrier.get("blocks_visibility") or barrier.get("locked") or not barrier.get("open")):
        return False, "barrier_blocks_visibility", barrier

    return False, "not_same_area", barrier


def can_hear_actor(
    scene_graph: Dict[str, Any],
    actor_id: str,
    target_actor_id: str,
    *,
    volume: str = "normal",
) -> Tuple[bool, str, Dict[str, Any]]:
    volume = _safe_str(volume).strip().lower() or "normal"
    a_area = actor_area(scene_graph, actor_id)
    t_area = actor_area(scene_graph, target_actor_id)
    if not a_area or not t_area:
        return False, "missing_spatial_reference", {}
    if a_area == t_area:
        return True, "same_area", {}

    barrier = barrier_between(scene_graph, a_area, t_area)
    if barrier:
        if volume == "whisper" and barrier.get("blocks_whisper"):
            return False, "barrier_blocks_whisper", barrier
        if volume in {"normal", "speak"} and barrier.get("muffles_sound"):
            return False, "barrier_muffles_sound", barrier
        if volume in {"yell", "shout"} and barrier.get("allows_yell"):
            return True, "yell_audible_through_barrier", barrier

    return False, "not_audible", barrier


def move_actor(scene_graph: Dict[str, Any], actor_id: str, to_area: str) -> Tuple[Dict[str, Any], Dict[str, Any]]:
    scene_graph = normalize_scene_graph(scene_graph)
    actor_id = _safe_str(actor_id).strip()
    to_area = _safe_str(to_area).strip()
    from_area = actor_area(scene_graph, actor_id)

    allowed, reason, barrier = can_move_between(scene_graph, from_area, to_area)
    if not allowed:
        return scene_graph, {
            "moved": False,
            "actor_id": actor_id,
            "from_area": from_area,
            "to_area": to_area,
            "reason": reason,
            "barrier": barrier,
        }

    actor_area_map = dict(_safe_dict(scene_graph.get("actor_area")))
    actor_area_map[actor_id] = to_area
    scene_graph["actor_area"] = actor_area_map

    areas = dict(_safe_dict(scene_graph.get("areas")))
    if from_area in areas:
        occupants = [x for x in _safe_list(areas[from_area].get("occupants")) if x != actor_id]
        areas[from_area]["occupants"] = occupants
    if to_area in areas:
        occupants = list(_safe_list(areas[to_area].get("occupants")))
        if actor_id not in occupants:
            occupants.append(actor_id)
        areas[to_area]["occupants"] = occupants
    scene_graph["areas"] = areas

    return scene_graph, {
        "moved": True,
        "actor_id": actor_id,
        "from_area": from_area,
        "to_area": to_area,
        "reason": "movement_allowed",
        "barrier": barrier,
    }

def _player_input_requests_spatial_movement(player_input: str) -> bool:
    text = _safe_str(player_input).strip().lower()
    return (
        "go to " in text
        or "move to " in text
        or "enter " in text
        or "walk to " in text
    )


def _target_area_from_player_input(player_input: str) -> str:
    text = _safe_str(player_input).strip().lower()
    aliases = {
        "cellar": "cellar",
        "basement": "cellar",
        "locked room": "locked_room",
        "room": "locked_room",
        "hallway": "hallway",
        "common room": "common_room",
        "tavern": "common_room",
    }
    for phrase, area_id in aliases.items():
        if phrase in text:
            return area_id
    return ""


def _player_input_requests_npc_contact_or_speech(player_input: str) -> bool:
    text = _safe_str(player_input).strip().lower()
    if "bran" not in text:
        return False
    return any(word in text for word in ["touch", "grab", "tap", "whisper", "say", "tell", "yell", "shout"])


def _npc_contact_or_speech_mode(player_input: str) -> str:
    text = _safe_str(player_input).strip().lower()
    if any(word in text for word in ["touch", "grab", "tap"]):
        return "touch"
    if "whisper" in text:
        return "whisper"
    if "yell" in text or "shout" in text:
        return "yell"
    return "speak"


def _fallback_spatial_narration(spatial_result: Dict[str, Any]) -> str:
    spatial_result = _safe_dict(spatial_result)
    reason = _safe_str(spatial_result.get("reason")).strip()
    mode = _safe_str(spatial_result.get("mode")).strip()
    target = _safe_str(spatial_result.get("target_actor_id") or spatial_result.get("target_area") or "there")

    if spatial_result.get("moved") is True:
        return f"Result: You move to {spatial_result.get('to_area')}."

    if spatial_result.get("allowed") is True:
        if mode == "yell":
            return f"Result: You yell to {target}."
        if mode == "speak":
            return f"Result: You speak to {target}."
        if mode == "whisper":
            return f"Result: You whisper to {target}."
        if mode == "touch":
            return f"Result: You touch {target}."
        return f"Result: {reason}"

    if reason == "barrier_locked":
        return "Result: The way is locked."
    if reason == "barrier_closed":
        return "Result: A closed barrier blocks the way."
    if reason == "barrier_blocks_touch":
        return f"Result: You cannot touch {target}; a barrier blocks you."
    if reason == "barrier_blocks_whisper":
        return f"Result: Your whisper does not reach {target}."
    if reason == "barrier_muffles_sound":
        return f"Result: Your voice is muffled by the barrier."
    if reason == "barrier_blocks_visibility":
        return f"Result: You cannot see {target} through the barrier."

    return f"Result: {reason}"

def _reconcile_spatial_action(
    final_result: Dict[str, Any],
    player_input: str,
) -> Dict[str, Any]:
    final_result = dict(_safe_dict(final_result))

    if not (
        _player_input_requests_spatial_movement(player_input)
        or _player_input_requests_npc_contact_or_speech(player_input)
    ):
        return final_result

    resolved_result = dict(_safe_dict(final_result.get("resolved_result")))
    visible_reason = _safe_str(
        resolved_result.get("visible_interaction_reason")
        or final_result.get("visible_interaction_reason")
    )
    if visible_reason.startswith("combat_"):
        return final_result

    session = _safe_dict(final_result.get("session"))
    simulation_state = _safe_dict(session.get("simulation_state") or final_result.get("simulation_state"))
    runtime_state = _safe_dict(session.get("runtime_state") or final_result.get("runtime_state"))

    if _player_input_requests_spatial_movement(player_input):
        target_area = _target_area_from_player_input(player_input)
        scene_graph = scene_graph_from_state(simulation_state)
        scene_graph, spatial_result = move_actor(scene_graph, "player", target_area)
        simulation_state = set_scene_graph(simulation_state, scene_graph)
        action_type = "move_area"
        visible = "spatial_moved" if spatial_result.get("moved") else f"spatial_{spatial_result.get('reason')}"
    else:
        mode = _npc_contact_or_speech_mode(player_input)
        scene_graph = scene_graph_from_state(simulation_state)
        if mode == "touch":
            allowed, reason, barrier = can_touch_actor(scene_graph, "player", "npc:bran")
        else:
            allowed, reason, barrier = can_hear_actor(scene_graph, "npc:bran", "player", volume=mode)
        spatial_result = {
            "allowed": allowed,
            "mode": mode,
            "actor_id": "player",
            "target_actor_id": "npc:bran",
            "reason": reason,
            "barrier": barrier,
        }
        action_type = "spatial_communication"
        visible = "spatial_communication_allowed" if spatial_result.get("allowed") else f"spatial_{spatial_result.get('reason')}"

    narration = _fallback_spatial_narration(spatial_result)

    resolved_result["action_type"] = action_type
    resolved_result["visible_interaction_reason"] = visible
    resolved_result["outcome"] = spatial_result.get("reason")
    resolved_result["spatial_result"] = spatial_result
    resolved_result["conversation_result"] = {
        "triggered": False,
        "reason": "spatial_action",
    }

    final_result["resolved_result"] = resolved_result
    final_result["spatial_result"] = spatial_result
    final_result["visible_interaction_reason"] = visible
    final_result["action_type"] = action_type
    final_result["outcome"] = spatial_result.get("reason")
    final_result["conversation_result"] = resolved_result["conversation_result"]
    final_result["narration"] = narration
    final_result["final_narration"] = narration
    final_result["summary"] = narration
    final_result["simulation_state"] = simulation_state

    if session:
        session["simulation_state"] = simulation_state
        session["runtime_state"] = runtime_state
        final_result["session"] = session

    result_obj = dict(_safe_dict(final_result.get("result")) or _safe_parse_mapping_payload(final_result.get("result")))
    if result_obj:
        result_obj["spatial_result"] = spatial_result
        result_obj["visible_interaction_reason"] = visible
        result_obj["action_type"] = action_type
        result_obj["outcome"] = spatial_result.get("reason")
        result_obj["conversation_result"] = resolved_result["conversation_result"]
        result_obj["narration"] = narration
        result_obj["final_narration"] = narration
        result_obj["summary"] = narration
        final_result["result"] = result_obj

    return final_result
