+def normalize_projection_to_evolution_signal(

+        return None, "unsupported_projection_kind"
+    if _contains_forbidden_claim(payload):

+        # keep evolution arcs NPC-scoped only.
+        return None, "npc_target_missing"
+    if not _npc_exists(simulation_state, npc_id):
+        return None, "npc_target_not_found"
+
+    summary = _projection_summary(projection)
+    if not summary:
+        return None, "summary_missing"
+
+    if kind == "relationship_delta":
+        try:

+        except Exception:
+            return None, "relationship_delta_not_integer"
+        delta = max(-2, min(2, delta))

+        except Exception:
+            importance = 0.5

+def ingest_evolution_signals(
+    *,

+    duplicates = 0

+            continue
+        signal_id = _safe_str(signal.get("signal_id"))

+            continue
+        if signal_id in existing_ids:
+            duplicates += 1
+            continue
+        existing_ids.add(signal_id)
+        existing.append(deepcopy(signal))
+        added += 1

+    state["signals"] = existing
+    state["summary"] = summarize_npc_evolution_state(runtime_state)

+def consume_evolution_signal(

+def consume_accepted_advisory_projections(
+    *,
+    runtime_state: Dict[str, Any],
+    simulation_state: Dict[str, Any],
+    turn_index: int,
+    max_signals_per_turn: int = 8,

+    """Convert accepted advisory projections into persistent NPC evolution arcs."""
+    runtime_state = runtime_state if isinstance(runtime_state, dict) else {}
+    simulation_state = _safe_dict(simulation_state)
+    advisory = _safe_dict(runtime_state.get("deferred_advisory"))
+    accepted = _safe_list(advisory.get("accepted"))

+    projection_decisions: List[Dict[str, Any]] = []
+    for accepted_row in accepted[-max_signals_per_turn:]:
+        projection = _safe_dict(_safe_dict(accepted_row).get("projection"))
+        if not projection:

+            simulation_state=simulation_state,
+            turn_index=int(turn_index),
+        )
+        if signal:
+            signals.append(signal)
+            projection_decisions.append(
+                {
+                    "projection_id": projection.get("candidate_id"),
+                    "status": "signal_created",
+                    "signal_id": signal.get("signal_id"),
+                    "kind": signal.get("kind"),
+                    "npc_id": signal.get("npc_id"),
+                }
+            )
+        else:

+                    "projection_id": projection.get("candidate_id"),

+                    "kind": projection.get("kind"),

+        "summary": summary,
