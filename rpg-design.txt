Bundle M52-M54 — Player Action Context + Suggested Actions v1
GOAL:
Create a deterministic player-facing action context that can power:
- UI suggested actions
- future autoplay player-agent prompts
- manual/debug inspection
- long-run campaign evaluation

This bundle adds:
- player action context runtime payload
- suggested action generation from current mode/state
- objective-aware suggestions
- NPC/social suggestions
- quest log/tracker suggestions
- story arc/campaign director suggestions
- combat/service/travel fallback suggestions
- bounded output
- player-visible only context
- manual tests with full debug

Important:
- Suggested actions are not authoritative outcomes.
- Suggested actions are player-facing affordances only.
- UI/LLM can choose an action, but simulation still resolves it.
- Hidden lore/secret state must not appear.
- The LLM player-agent should eventually receive this payload instead of raw simulation state.

+    simulation_state: Dict[str, Any],
+    *,

+    npcs = _visible_npcs(simulation_state)
+    location = _derive_location(simulation_state)
+    mode = _derive_mode(simulation_state)
+    arcs = _active_arc_rows(recap)

+    for row in _objective_actions(objectives):
+        actions.append(row)
+    for row in _npc_actions(npcs):
+        actions.append(row)
+    for row in _arc_actions(arcs):
+        actions.append(row)
+    for row in _mode_actions(mode, location):
+        actions.append(row)
+
+    actions.sort(
+        key=lambda row: (
+            -int(row.get("priority") or 0),
+            str(row.get("category") or ""),
+            str(row.get("action_id") or ""),
+        )
+    )
+    return actions[:limit]

+from app.rpg.player_action_context.runtime import build_player_action_context
+from app.rpg.session_store import get_session

+
+router = APIRouter(prefix="/api/rpg/player_action_context", tags=["rpg-player-action-context"])
+
+
+class PlayerActionContextRequest(BaseModel):
+    session_id: str
+    turn_index: int = 0
+    limit: int = Field(default=12, ge=0, le=12)
+
+
+def _simulation_state_for_session(session_id: str) -> Dict[str, Any]:
+    session = get_session(session_id)
+    if not isinstance(session, dict):
+        raise KeyError(f"session_not_found:{session_id}")
+    return session.setdefault("simulation_state", {})
+
+
+@router.post("/payload")
+def player_action_context_payload(request: PlayerActionContextRequest) -> Dict[str, Any]:
+    simulation_state = _simulation_state_for_session(request.session_id)
+    return build_player_action_context(
+        simulation_state,
+        turn_index=request.turn_index,
+        limit=request.limit,
+    )

+from tests.rpg.manual.scenarios.registry import build_service_scenarios

+def _m49_m51_filter(check):
+    check_type = str(check.get("type") or "")
+    return (
+        check_type.startswith("quest_log")
+        or check_type.startswith("objective_tracker")
+        or check_type == "campaign_recap_objective_tracker"
+    )
+
+
+def _m52_m54_filter(check):
+    check_type = str(check.get("type") or "")
+    return (
+        check_type.startswith("player_action_context")
+        or check_type == "suggested_actions"
+    )
+
+
+def test_player_action_context_checks_do_not_route_to_m49_m51():
+    scenarios = build_service_scenarios()
+    names = [
+        "player_action_context_includes_active_objectives",
+        "suggested_actions_prioritize_pinned_objective",
+        "player_action_context_includes_nearby_npc_social_actions",
+        "player_action_context_combat_mode_suggests_combat_actions",

+        "player_action_context_excludes_unrevealed_secret_lore",
+        "player_action_context_output_is_bounded",
+        "player_action_context_contains_player_agent_schema",
+    ]
+
+    for name in names:
+        checks = scenarios[name]["checks"]
+        m52 = [check for check in checks if isinstance(check, dict) and _m52_m54_filter(check)]
+        bad_m49 = [check for check in checks if isinstance(check, dict) and _m49_m51_filter(check)]
